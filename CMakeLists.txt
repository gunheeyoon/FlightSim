cmake_minimum_required(VERSION 3.16)
project(AircraftPrototype LANGUAGES C CXX)
find_package(Threads REQUIRED)
add_executable(aircraft_server src/main.cpp)
target_compile_features(aircraft_server PRIVATE cxx_std_17)
target_link_libraries(aircraft_server PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(aircraft_server PRIVATE ws2_32)
endif()
add_executable(flight_simulator simulator/main.c simulator/route.c)
target_compile_features(flight_simulator PRIVATE c_std_11)
if(WIN32)
  target_link_libraries(flight_simulator PRIVATE ws2_32)
else()
  target_link_libraries(flight_simulator PRIVATE m)
endif()
if(MSVC)
  target_compile_options(flight_simulator PRIVATE /W4)
  target_compile_definitions(flight_simulator PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(flight_simulator PRIVATE -Wall -Wextra -Wpedantic)
endif()
configure_file(routes/seattle_short.json routes/seattle_short.json COPYONLY)
if(MSVC)
  target_compile_options(aircraft_server PRIVATE /W4 /permissive-)
else()
  target_compile_options(aircraft_server PRIVATE -Wall -Wextra -Wpedantic)
endif()
